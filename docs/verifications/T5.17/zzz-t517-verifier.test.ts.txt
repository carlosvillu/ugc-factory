// VERIFIER T5.17 — test INDEPENDIENTE del verifier (no del implementer). Cubre las causas que la
// Verificación nombra pero que el test de integración del implementer NO conduce end-to-end: 401 «invalid
// api key» (credencial) y 422 validación (array). Ambas por el CAMINO REAL del consumer, asertando contra
// el jsonb crudo de `step_run.error` leído de la BD (no el message del assert). Fixtures elegidos por el
// verifier. Se ejecuta y se BORRA; queda copia en docs/verifications/T5.17/.
import { afterAll, afterEach, beforeAll, beforeEach, describe, expect, it } from 'vitest';
import { createRun } from '@ugc/core/orchestrator';
import type { StepExecutor } from '@ugc/core/orchestrator';
import { makeFalClient } from '@ugc/core/generation';
import { stepExecuteJob } from '@ugc/core/jobs';
import { createTestDatabase } from '@ugc/test-utils';
import type { TestDatabase } from '@ugc/test-utils';
import { PgBoss } from 'pg-boss';
import { writeFileSync } from 'node:fs';

import { seedProject, startWorkerWith, stopBossAndWait, waitFor } from '../helpers';
import { runGenerationStep } from '../../src/executors/_shared';

const ENDPOINT = 'fal-ai/flux-2';
const DUMP = '/Users/carlosvillu/Developer/__GARBAGE__/UGC_Ads/docs/verifications/T5.17/03-raw-jsonb-dump.txt';

let tdb: TestDatabase;
let dump = '';

async function fetchStep(runId: string): Promise<{ status: string; error: unknown } | undefined> {
  const { rows } = await tdb.pool.query<{ status: string; error: unknown }>(
    'SELECT status, error FROM step_run WHERE run_id = $1 ORDER BY id LIMIT 1',
    [runId],
  );
  const r = rows[0];
  return r ? { status: r.status, error: r.error } : undefined;
}

beforeAll(async () => {
  tdb = await createTestDatabase({ label: 'worker:zzz-t517-verifier' });
  const seedBoss = new PgBoss({ connectionString: tdb.connectionString, max: 2 });
  seedBoss.on('error', () => undefined);
  await seedBoss.start();
  await stopBossAndWait(seedBoss);
});

afterAll(async () => {
  writeFileSync(DUMP, dump);
  await tdb.close();
});

beforeEach(async () => {
  await tdb.pool.query('TRUNCATE step_run, pipeline_run, project CASCADE');
  await tdb.pool.query('DELETE FROM pgboss.job WHERE name = $1', [stepExecuteJob.name]);
});

let harnessCleanup: (() => Promise<void>) | undefined;
afterEach(async () => {
  if (harnessCleanup !== undefined) {
    await harnessCleanup();
    harnessCleanup = undefined;
  }
});

function providerErrorExecutor(status: number, statusText: string, body: unknown): StepExecutor {
  const falFetch: typeof globalThis.fetch = () =>
    Promise.resolve(
      new Response(JSON.stringify(body), {
        status,
        statusText,
        headers: { 'content-type': 'application/json' },
      }),
    );
  const fal = makeFalClient({
    credentials: 'test-fal-key-not-a-secret',
    fetch: falFetch,
    sleep: () => Promise.resolve(),
    pollIntervalMs: 0,
    maxRetries: 0,
  });
  return async () => {
    await runGenerationStep(() => fal.submit(ENDPOINT, { prompt: 'x' }));
  };
}

async function launchN7bWith(executor: StepExecutor, maxRetries: number): Promise<string> {
  const { deps, cleanup } = await startWorkerWith(tdb, { N7b: executor }, { retryBackoffMs: 50 });
  harnessCleanup = cleanup;
  const projectId = await seedProject(tdb);
  const { runId } = await createRun(deps, {
    projectId,
    nodes: [{ key: 'N7b', nodeKey: 'N7b', variantId: 'v1', dependsOn: [], config: {}, maxRetries }],
  });
  return runId;
}

interface PersistedStepError {
  message?: string;
  permanent?: boolean;
  status?: number;
  detail?: string;
}

describe('VERIFIER T5.17 — causas 401/422 end-to-end en step_run.error (jsonb crudo)', () => {
  it('401 «invalid api key» (credencial) → step_run.error.detail contiene la causa; permanente en 1 submit', async () => {
    const runId = await launchN7bWith(
      providerErrorExecutor(401, 'Unauthorized', { detail: 'Invalid API key provided.' }),
      81,
    );
    await waitFor(async () => (await fetchStep(runId))?.status === 'failed', 30_000, '401 failed', 50);
    const step = await fetchStep(runId);
    const err = step?.error as PersistedStepError;
    dump += `\n=== 401 credencial ===\nstatus=${step?.status}\nerror=${JSON.stringify(step?.error, null, 2)}\n`;
    expect(step?.status).toBe('failed');
    expect(err.permanent).toBe(true);
    expect(err.status).toBe(401);
    expect(err.detail).toBeDefined();
    expect(err.detail).toContain('Invalid API key');
    // El message NO contuvo la causa de contrabando (statusText = 'Unauthorized'); solo el detail la trae
    // (y el _shared re-inyecta «Causa (fal):» en el message). El fixture no es verde-decorativo.
    expect(err.message).toMatch(/401: Unauthorized/);
  }, 60_000);

  it('422 validación (array de objetos) → detail serializa el array entero, no [object Object]', async () => {
    const runId = await launchN7bWith(
      providerErrorExecutor(422, 'Unprocessable Entity', {
        detail: [{ loc: ['body', 'voice_id'], msg: 'invalid voice id', type: 'value_error' }],
      }),
      81,
    );
    await waitFor(async () => (await fetchStep(runId))?.status === 'failed', 30_000, '422 failed', 50);
    const step = await fetchStep(runId);
    const err = step?.error as PersistedStepError;
    dump += `\n=== 422 validación ===\nstatus=${step?.status}\nerror=${JSON.stringify(step?.error, null, 2)}\n`;
    expect(step?.status).toBe('failed');
    expect(err.permanent).toBe(true);
    expect(err.status).toBe(422);
    expect(err.detail).toBeDefined();
    expect(err.detail).toContain('invalid voice id');
    expect(err.detail).not.toContain('[object Object]');
  }, 60_000);

  it('422 con signed URL reflejada → el sig NO se persiste en el jsonb (redacción end-to-end)', async () => {
    const runId = await launchN7bWith(
      providerErrorExecutor(422, 'Unprocessable Entity', {
        detail: [
          {
            loc: ['body', 'audio_url'],
            msg: 'invalid audio at https://fal.media/files/x.mp3?sig=SECRET123&exp=999',
            type: 'value_error',
          },
        ],
      }),
      81,
    );
    await waitFor(async () => (await fetchStep(runId))?.status === 'failed', 30_000, '422 sig failed', 50);
    const step = await fetchStep(runId);
    const raw = JSON.stringify(step?.error);
    dump += `\n=== 422 signed-url redaction ===\nrawJsonb=${raw}\n`;
    // El secreto NO aparece en NINGUNA parte del jsonb persistido (ni detail ni message).
    expect(raw).not.toContain('SECRET123');
    expect(raw).toContain('fal.media/files/x.mp3');
    expect(raw).toContain('?<redacted>');
  }, 60_000);
});
