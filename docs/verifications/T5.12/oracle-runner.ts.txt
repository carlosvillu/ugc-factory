let fails = 0, passes = 0;
const stack: string[] = [];
export function describe(n: string, f: () => void) { stack.push(n); f(); stack.pop(); }
export function it(n: string, f: () => void) {
  try { f(); passes++; console.log(`  PASS  ${n}`); }
  catch (e) { fails++; console.log(`  FAIL  ${n}\n        ${(e as Error).message.split('\n')[0]}`); }
}
export function expect(actual: unknown, msg?: string) {
  const j = (v: unknown) => JSON.stringify(v);
  const api = {
    toBe: (e: unknown) => { if (!Object.is(actual, e)) throw new Error(`${msg ?? ''} expected ${j(e)} got ${j(actual)}`); },
    toEqual: (e: unknown) => { if (j(actual) !== j(e)) throw new Error(`${msg ?? ''} expected ${j(e)} got ${j(actual)}`); },
    toBeUndefined: () => { if (actual !== undefined) throw new Error(`${msg ?? ''} expected undefined got ${j(actual)}`); },
    toBeDefined: () => { if (actual === undefined) throw new Error(`${msg ?? ''} expected defined`); },
    toBeTruthy: () => { if (!actual) throw new Error(`${msg ?? ''} expected truthy got ${j(actual)}`); },
    toBeGreaterThan: (e: number) => { if (!((actual as number) > e)) throw new Error(`${msg ?? ''} expected >${e} got ${j(actual)}`); },
  };
  return api;
}
export function summary() { console.log(`\n=== RESULT: ${passes} passed, ${fails} failed ===`); if (fails > 0) process.exitCode = 1; }
